在Python 3.8.1与PySide2环境下，为文本展示区域（如QPlainTextEdit或QTextEdit）实现一个能够跟随文本内容同步滚动的行号显示功能，是提升代码编辑器或日志查看器等应用用户体验的关键。这通常需要创建一个自定义的行号部件，并将其与主文本编辑器的视口和滚动条进行动态绑定。以下将结合搜索结果，为您提供一个清晰、可行的实现方案。
核心思路与架构
实现该功能的核心在于创建一个独立的、用于显示行号的部件（例如QWidget），将其放置在文本编辑器的左侧。该部件需要实时响应两个主要事件：1) 文本编辑器内容的更新（行数变化）；2) 文本编辑器视口的滚动。通过同步两者的视口和内容，即可实现行号跟随文本滚动的效果。
一个稳健的实现通常包含以下组件：

主文本编辑器：如QPlainTextEdit，负责显示和编辑文本内容。
自定义行号区域：一个继承自QWidget的类，负责绘制行号。
同步机制：通过信号和槽（Signal & Slot）连接文本编辑器的updateRequest信号（当视口更新或内容变化时触发）和垂直滚动条的valueChanged信号，来驱动行号区域的更新和滚动同步。

详细实现步骤
以下步骤将引导您完成一个完整的、可运行的示例。
步骤一：创建自定义行号区域类
这个类负责计算并绘制当前可见区域的行号。其关键点在于根据文本编辑器的视口位置和行高，计算出需要显示的行号范围。
from PySide2.QtWidgets import QWidget, QPlainTextEdit, QApplication, QVBoxLayout, QHBoxLayout
from PySide2.QtGui import QPainter, QFont, QColor
from PySide2.QtCore import Qt, QRect, QSize

class LineNumberArea(QWidget):
    def __init__(self, editor):
        super().__init__(editor)
        self._editor = editor
        # 设置行号区域的初始宽度，可根据最大行号位数调整
        self.update_width()

    def update_width(self):
        # 根据编辑器可能的最大行号位数，动态计算并设置行号区域的宽度
        digits = len(str(max(1, self._editor.blockCount())))
        space = 10 + self.fontMetrics().width('9') * digits
        self.setFixedWidth(space)

    def sizeHint(self):
        return QSize(self.width(), 0)

    def paintEvent(self, event):
        # 绘制行号
        painter = QPainter(self)
        painter.fillRect(event.rect(), QColor(240, 240, 240))  # 设置行号区域背景色

        # 获取当前可见的第一个文本块（行）
        block = self._editor.firstVisibleBlock()
        block_number = block.blockNumber()
        top = self._editor.blockBoundingGeometry(block).translated(self._editor.contentOffset()).top()
        bottom = top + self._editor.blockBoundingRect(block).height()

        # 遍历所有当前可见的文本块并绘制行号
        while block.isValid() and top <= event.rect().bottom():
            if block.isVisible() and bottom >= event.rect().top():
                number = str(block_number + 1)
                painter.setPen(Qt.black)
                painter.drawText(0, int(top), self.width() - 5, self.fontMetrics().height(),
                                 Qt.AlignRight, number)

            block = block.next()
            top = bottom
            bottom = top + self._editor.blockBoundingRect(block).height()
            block_number += 1

步骤二：创建主文本编辑器类并集成行号区域
这个类将主文本编辑器与行号区域组合在一起，并建立它们之间的连接。
class CodeEditor(QPlainTextEdit):
    def __init__(self):
        super().__init__()
        self.line_number_area = LineNumberArea(self)

        # 连接信号与槽
        self.blockCountChanged.connect(self.update_line_number_area_width)
        self.updateRequest.connect(self.update_line_number_area)
        self.verticalScrollBar().valueChanged.connect(self.sync_scroll)

        # 初始更新
        self.update_line_number_area_width()

    def update_line_number_area_width(self):
        # 当文档行数变化时，更新行号区域的宽度并设置编辑器左边距
        self.setViewportMargins(self.line_number_area.sizeHint().width(), 0, 0, 0)
        self.line_number_area.update_width()

    def update_line_number_area(self, rect, dy):
        # 当编辑器视口更新（如滚动、内容变化）时，重绘行号区域
        if dy:
            self.line_number_area.scroll(0, dy)
        else:
            self.line_number_area.update(0, rect.y(), self.line_number_area.width(), rect.height())

    def sync_scroll(self):
        # 确保行号区域的垂直滚动与编辑器同步
        self.line_number_area.update()

    def resizeEvent(self, event):
        super().resizeEvent(event)
        # 调整行号区域的大小和位置，使其始终覆盖编辑器左侧区域
        cr = self.contentsRect()
        self.line_number_area.setGeometry(QRect(cr.left(), cr.top(), self.line_number_area.width(), cr.height()))

步骤三：在应用程序窗口中部署
最后，将自定义的编辑器放入主窗口进行测试。
class MainWindow(QWidget):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("PySide2 带行号的文本编辑器")
        self.setGeometry(300, 300, 800, 600)

        layout = QVBoxLayout(self)
        self.editor = CodeEditor()
        layout.addWidget(self.editor)

        # 加载示例文本或由用户输入
        self.editor.setPlainText("\n".join(f"这是第 {i+1} 行文本。" for i in range(100)))

if __name__ == '__main__':
    app = QApplication([])
    window = MainWindow()
    window.show()
    app.exec_()

关键点与优化建议

性能考虑：当处理极大文本（如数万行）时，频繁的绘制和信号传递可能影响性能。可以优化paintEvent，只绘制当前可见区域的行号，并考虑对行号进行缓存。
滚动同步精度：上述方法通过updateRequest和滚动条信号实现了基础同步。对于更精确的同步，特别是快速滚动时，可以借鉴直接连接两个滚动条valueChanged信号的方法，但需注意避免信号循环。
外观定制：您可以轻松修改LineNumberArea.paintEvent中的背景色、字体颜色和对齐方式，以匹配应用程序的主题。
原理延伸：此方案的核心思想与在QTableView中通过重写headerData方法显示行号有异曲同工之妙，都是通过一个独立的视图组件来展示元信息（行号）。

通过以上步骤，您可以在PySide2应用中实现一个专业且响应灵敏的带滚动行号的文本编辑器。此方案结构清晰，分离了显示逻辑与编辑逻辑，便于后续的功能扩展和维护。
